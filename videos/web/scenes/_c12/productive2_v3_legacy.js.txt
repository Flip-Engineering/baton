const NOISE = `
float h13(vec3 p){ p = fract(p*.1031); p += dot(p, p.zyx+31.32); return fract((p.x+p.y)*p.z); }
float vn3(vec3 x){ vec3 i = floor(x), f = fract(x); f = f*f*(3.-2.*f);
  return mix(mix(mix(h13(i), h13(i+vec3(1,0,0)), f.x), mix(h13(i+vec3(0,1,0)), h13(i+vec3(1,1,0)), f.x), f.y),
             mix(mix(h13(i+vec3(0,0,1)), h13(i+vec3(1,0,1)), f.x), mix(h13(i+vec3(0,1,1)), h13(i+vec3(1,1,1)), f.x), f.y), f.z); }
float fbm3(vec3 p){ float v = 0., a = .5; for(int i=0;i<4;i++){ v += a*vn3(p); p = p*2.07+vec3(1.7,9.2,3.1); a *= .5; } return v; }
`;
const TM_VS = `
attribute float aT;
varying vec3 vW; varying vec3 vN; varying float vT; varying vec2 vUv; varying vec3 vI;
void main(){
  mat4 m = modelMatrix;
#ifdef USE_INSTANCING
  m = modelMatrix * instanceMatrix;
#endif
  vec4 w = m * vec4(position, 1.0);
  vW = w.xyz; vN = normalize(mat3(m) * normal); vUv = uv; vT = aT; vI = vec3(0.);
#ifdef USE_INSTANCING_COLOR
  vI = instanceColor;
#endif
  gl_Position = projectionMatrix * viewMatrix * w;
}`;
const TM_FS = NOISE + `
uniform float uBase, uAmp, uScale, uFlow, uEps, uAir, uFog, uTime, uMapAmt, uSky, uHot;
uniform vec4 uGrad, uStripe, uStripe2, uSun, uFlowDir;
uniform vec3 uSP[4]; uniform vec4 uS[4];
uniform sampler2D uMap;
varying vec3 vW; varying vec3 vN; varying float vT; varying vec2 vUv; varying vec3 vI;
void main(){
  vec3 N = normalize(vN); if(!gl_FrontFacing) N = -N;
  vec3 V = normalize(cameraPosition - vW);
  float T = uBase + dot(vW, uGrad.xyz)*uGrad.w;
#ifdef HUMAN
  float fr = vT;
  if(fr > 15.){ if(vI.b < .5) discard; fr = .58; }
  else if(fr > 5.) fr = mix(.9, .6, step(.5, vI.b));
  T += vI.r * fr * (1. + (fbm3(vW*7.) - .5)*.35*(1. - step(.97, fr)));
#else
  T += vT + vI.r;
#endif
  if(uAmp != 0.) T += (fbm3(vW*uScale + uFlowDir.xyz*uTime*uFlow) - .5)*uAmp*(1. + vI.g);
  if(uStripe.w != 0.) T += sin(dot(vW, uStripe.xyz))*uStripe.w;
  if(uStripe2.w != 0.) T += smoothstep(.82, .9, fract(dot(vW, uStripe2.xyz)))*uStripe2.w;
  if(uMapAmt != 0.) T += texture2D(uMap, vUv).a * uMapAmt;
  float spec = 0.;
  vec3 R = reflect(-V, N);
  for(int i=0;i<4;i++){
    if(uS[i].x == 0.) continue;
    vec3 d = uSP[i] - vW; float r = max(length(d), .05); vec3 L = d/r;
    float fall = 1./(1. + r*r/(uS[i].y*uS[i].y));
    T += uS[i].x*uS[i].z*max(0., dot(N, L)*.8+.2)*fall;
    spec += uS[i].x*uS[i].w*pow(max(0., dot(R, L)), 18.)*fall;
  }
  T += uSun.w*max(0., dot(N, uSun.xyz));
  T -= uSky*max(0., N.y);
  float fres = pow(1. - abs(dot(N, V)), 3.);
  float eps = uEps*(1. - .5*fres);
  T = eps*T + (1. - eps)*uAir + spec*(1. - eps);
  T *= uHot;
  float dist = length(cameraPosition - vW);
  T = mix(T, uAir, 1. - exp(-dist*uFog));
  gl_FragColor = vec4(T, 0., 0., 1.);
}`;

// A "set" is one place in the plant: its own scene, camera and shared uniforms (time, air, heat sources).
function makeSet(name) {
  const U = { uTime: { value: 0 }, uAir: { value: AIR }, uFog: { value: 0.004 }, uSky: { value: 0.004 }, uHot: { value: 1 },
    uSun: { value: new THREE.Vector4(0.3, 0.8, 0.5, 0) },
    uSP: { value: [0, 1, 2, 3].map(() => new THREE.Vector3()) }, uS: { value: [0, 1, 2, 3].map(() => new THREE.Vector4()) } };
  const scene = new THREE.Scene();
  const cam = new THREE.PerspectiveCamera(45, W / H, 0.05, 4000);
  return { name, U, scene, cam, mats: [] };
}
const blank = new THREE.DataTexture(new Uint8Array([0, 0, 0, 0]), 1, 1); blank.needsUpdate = true;
function mat(set, o = {}) {
  const m = new THREE.ShaderMaterial({
    vertexShader: TM_VS, fragmentShader: TM_FS, side: o.side ?? THREE.DoubleSide,
    defines: o.human ? { HUMAN: 1 } : {},
    uniforms: { ...set.U,
      uBase: { value: o.base ?? AIR }, uAmp: { value: o.amp ?? 0.004 }, uScale: { value: o.scale ?? 1 }, uFlow: { value: o.flow ?? 0 },
      uFlowDir: { value: new THREE.Vector4(...(o.flowDir ?? [0, -1, 0]), 0) }, uEps: { value: o.eps ?? 0.92 },
      uGrad: { value: new THREE.Vector4(...(o.grad ?? [0, 1, 0, 0])) }, uStripe: { value: new THREE.Vector4(...(o.stripe ?? [0, 0, 0, 0])) },
      uStripe2: { value: new THREE.Vector4(...(o.stripe2 ?? [0, 0, 0, 0])) },
      uMap: { value: o.map ?? blank }, uMapAmt: { value: o.mapAmt ?? 0 } },
  });
  set.mats.push(m);
  return m;
}
function heat(set, i, p, temp, radius, absorb = 0.05, spec = 0.6) { set.U.uSP.value[i].set(...p); set.U.uS.value[i].set(temp, radius, absorb, spec); }
function add(set, geo, m, x = 0, y = 0, z = 0, ry = 0, parent = null) {
  const o = new THREE.Mesh(geo, m); o.position.set(x, y, z); o.rotation.y = ry; (parent || set.scene).add(o); return o;
}
// per-vertex temperature attribute
function withT(geo, fn) {
  const p = geo.attributes.position, n = geo.attributes.normal, a = new Float32Array(p.count);
  for (let i = 0; i < p.count; i++) a[i] = fn(p.getX(i), p.getY(i), p.getZ(i), n ? n.getX(i) : 0, n ? n.getY(i) : 0, n ? n.getZ(i) : 0);
  geo.setAttribute('aT', new THREE.BufferAttribute(a, 1));
  return geo;
}
const constT = (geo, v) => withT(geo, () => v);
function tf(geo, { x = 0, y = 0, z = 0, rx = 0, ry = 0, rz = 0, sx = 1, sy = 1, sz = 1 } = {}) {
  const m = new THREE.Matrix4().compose(new THREE.Vector3(x, y, z), new THREE.Quaternion().setFromEuler(new THREE.Euler(rx, ry, rz)), new THREE.Vector3(sx, sy, sz));
  return geo.applyMatrix4(m);
}
const merge = (gs) => mergeGeometries(gs.map(g => g.index ? g.toNonIndexed() : g), false);

// ---------------------------------------------------------------- shapes
function ibeam(h, w = 0.4, d = 0.4, tw = 0.05) {           // an H column, height h, standing on y=0
  const s = new THREE.Shape();
  s.moveTo(-w / 2, -d / 2); s.lineTo(w / 2, -d / 2); s.lineTo(w / 2, -d / 2 + tw); s.lineTo(tw / 2, -d / 2 + tw); s.lineTo(tw / 2, d / 2 - tw);
  s.lineTo(w / 2, d / 2 - tw); s.lineTo(w / 2, d / 2); s.lineTo(-w / 2, d / 2); s.lineTo(-w / 2, d / 2 - tw); s.lineTo(-tw / 2, d / 2 - tw);
  s.lineTo(-tw / 2, -d / 2 + tw); s.lineTo(-w / 2, -d / 2 + tw); s.closePath();
  const g = new THREE.ExtrudeGeometry(s, { depth: h, bevelEnabled: false });
  return tf(g, { rx: -Math.PI / 2 });
}
const box = (w, h, d) => new THREE.BoxGeometry(w, h, d);
const cyl = (r0, r1, h, n = 16, open = false) => new THREE.CylinderGeometry(r0, r1, h, n, 1, open);
function lathe(pts, n = 40) { return new THREE.LatheGeometry(pts.map(([r, y]) => new THREE.Vector2(r, y)), n); }
// a lattice mast / truss from thin bars between points
function bars(list, r = 0.06) {
  const gs = [];
  for (const [a, b] of list) {
    const A = new THREE.Vector3(...a), Bv = new THREE.Vector3(...b), d = Bv.clone().sub(A), L = d.length();
    const g = new THREE.CylinderGeometry(r, r, L, 5, 1, true);
    g.applyMatrix4(new THREE.Matrix4().makeTranslation(0, L / 2, 0));
    const q = new THREE.Quaternion().setFromUnitVectors(new THREE.Vector3(0, 1, 0), d.normalize());
    g.applyMatrix4(new THREE.Matrix4().compose(A, q, new THREE.Vector3(1, 1, 1)));
    gs.push(g);
  }
  return merge(gs);
}
function mast(h, w, levels) {                                // a square lattice tower (pylons, gantries)
  const L = [];
  for (let k = 0; k < levels; k++) {
    const y0 = h * k / levels, y1 = h * (k + 1) / levels, w0 = lerp(w, w * 0.45, k / levels), w1 = lerp(w, w * 0.45, (k + 1) / levels);
    const c0 = [[-w0, -w0], [w0, -w0], [w0, w0], [-w0, w0]], c1 = [[-w1, -w1], [w1, -w1], [w1, w1], [-w1, w1]];
    for (let i = 0; i < 4; i++) {
      const j = (i + 1) % 4;
      L.push([[c0[i][0], y0, c0[i][1]], [c1[i][0], y1, c1[i][1]]]);
      L.push([[c0[i][0], y0, c0[i][1]], [c1[j][0], y1, c1[j][1]]]);
      L.push([[c1[i][0], y1, c1[i][1]], [c1[j][0], y1, c1[j][1]]]);
    }
  }
  return bars(L, 0.05 * w);
}

function humanGeo(pose) {
  const gs = [];
  const put = (g, M, f) => { g.applyMatrix4(M); constT(g, f); gs.push(g.index ? g.toNonIndexed() : g); };
  const M = (x, y, z, rx = 0, ry = 0, rz = 0) => new THREE.Matrix4().compose(new THREE.Vector3(x, y, z), new THREE.Quaternion().setFromEuler(new THREE.Euler(rx, ry, rz, 'XYZ')), new THREE.Vector3(1, 1, 1));
  const bob = pose.bob || 0, lean = pose.lean || 0;
  const root = M(0, bob, 0, lean);
  // torso (jacket) and hips
  const torso = lathe([[0.001, 0.86], [0.15, 0.88], [0.165, 1.0], [0.155, 1.1], [0.175, 1.24], [0.205, 1.36], [0.19, 1.44], [0.12, 1.49], [0.05, 1.51]], 14);
  torso.scale(1, 1, 0.62);
  put(torso, root, pose.coat ?? 0.7);
  // hi-vis vest band: a cooler stripe (reflective tape)
  put(tf(lathe([[0.18, 1.18], [0.182, 1.22]], 14).scale(1, 1, 0.64), {}), root, 0.5);
  // neck, head (face hotter at the front), hard hat
  put(cyl(0.055, 0.06, 0.12, 8), root.clone().multiply(M(0, 1.54, 0.01)), 0.98);
  const head = new THREE.SphereGeometry(0.105, 16, 12); head.scale(0.92, 1.12, 1.0);
  head.applyMatrix4(root.clone().multiply(M(0, 1.67, 0.015)));
  withT(head, (x, y, z, nx, ny, nz) => 0.9 + 0.08 * Math.max(0, nz) - 0.05 * Math.max(0, ny) + (nz > 0.72 && ny > 0.05 && ny < 0.35 && Math.abs(nx) > 0.12 && Math.abs(nx) < 0.42 ? 0.1 : 0) + (nz > 0.8 && ny < -0.3 && ny > -0.55 ? 0.05 : 0));
  gs.push(head.toNonIndexed());
  const nose = new THREE.SphereGeometry(0.022, 6, 5); nose.scale(0.8, 1.3, 1.1);
  put(nose, root.clone().multiply(M(0, 1.655, 0.118)), 0.86);
  const hat = new THREE.SphereGeometry(0.128, 14, 7, 0, TAU, 0, Math.PI / 2); hat.scale(1, 0.9, 1.1);
  put(hat, root.clone().multiply(M(0, 1.72, 0.0)), 10.6);
  put(cyl(0.16, 0.16, 0.015, 16).scale(1, 1, 1.2), root.clone().multiply(M(0, 1.725, 0.03)), 20.58);
  for (const sx of [-1, 1]) { const ear = new THREE.SphereGeometry(0.022, 6, 5); ear.scale(0.5, 1.3, 1); put(ear, root.clone().multiply(M(sx * 0.1, 1.665, 0.0)), 0.88); }
  // legs: thigh swing a, knee bend k (foot goes back)
  for (const [side, a, k] of [[-1, pose.lA, pose.lK], [1, pose.rA, pose.rK]]) {
    const hip = root.clone().multiply(M(side * 0.095, 0.9, 0, -a));
    put(tf(cyl(0.085, 0.07, 0.46, 10), { y: -0.23 }), hip, 0.66);
    const knee = hip.clone().multiply(M(0, -0.46, 0, k));
    put(tf(cyl(0.065, 0.05, 0.44, 10), { y: -0.22 }), knee, 0.64);
    put(tf(box(0.11, 0.09, 0.27), { y: -0.47, z: 0.05 }), knee, 0.55);
  }
  // arms: swing a, elbow e (forearm forward), raise r
  for (const [side, a, e] of [[-1, pose.lArm, pose.lE], [1, pose.rArm, pose.rE]]) {
    const sh = root.clone().multiply(M(side * 0.215, 1.41, 0, -a, 0, side * (pose.spread || 0.06)));
    put(tf(cyl(0.062, 0.05, 0.3, 8), { y: -0.15 }), sh, 0.7);
    const el = sh.clone().multiply(M(0, -0.3, 0, -e));
    put(tf(cyl(0.048, 0.04, 0.27, 8), { y: -0.135 }), el, 0.76);
    const hand = new THREE.SphereGeometry(0.048, 8, 6); hand.scale(0.75, 1.25, 0.95);
    put(hand, el.clone().multiply(M(0, -0.31, 0)), 1.0);
  }
  if (pose.pole) put(tf(cyl(0.02, 0.02, 3.2, 6), { y: 0, rx: 1.1 }), root.clone().multiply(M(0.2, 1.0, 0.35)), 0.55);
  return mergeGeometries(gs, false);
}

class Crowd {
  constructor(set, max, o = {}) {
    this.m = mat(set, { human: 1, base: 0, amp: 0.003, scale: 6, eps: 0.97, ...o });
    this.meshes = POSES.map(g => { const im = new THREE.InstancedMesh(g, this.m, max); im.frustumCulled = false;
      for (let i = 0; i < max; i++) im.setColorAt(i, new THREE.Color(0, 0, 0)); im.count = 0; set.scene.add(im); return im; });
    this.q = new THREE.Quaternion(); this.v = new THREE.Vector3(); this.s = new THREE.Vector3(); this.mm = new THREE.Matrix4(); this.c = new THREE.Color();
  }
  begin() { for (const im of this.meshes) im.count = 0; }
  put(pose, x, y, z, ry, sc, temp, rough = 0, hat = 1, wide = 1) {
    const im = this.meshes[pose], i = im.count++;
    this.q.setFromAxisAngle(new THREE.Vector3(0, 1, 0), ry);
    this.mm.compose(this.v.set(x, y, z), this.q, this.s.set(sc * wide, sc, sc * wide));
    im.setMatrixAt(i, this.mm); im.setColorAt(i, this.c.setRGB(temp, rough, hat));
  }
  end() { for (const im of this.meshes) { im.instanceMatrix.needsUpdate = true; if (im.instanceColor) im.instanceColor.needsUpdate = true; } }
}

// ---------------------------------------------------------------- particles: stateless streaks (sparks, embers, fume)
// Each particle is a pure function of (seed, t): born on a loop of period P, flies ballistically, cools as it flies.
const STREAK_VS = NOISE + `
attribute vec2 corner; attribute vec4 seed;
uniform float uTime, uPeriod, uLife, uStreak, uWidth, uT0, uCool, uSpread, uRate, uUp, uRise, uGrow;
uniform vec3 uO, uV, uG, uBox;
uniform vec2 uRes;
varying float vT; varying vec2 vC; varying float vA;
vec3 P(float a, vec3 o, vec3 v){ return o + v*a + .5*uG*a*a + (uRise>0. ? vec3(fbm3(vec3(seed.xy*9., a*.7))-.5, 0., fbm3(vec3(seed.zw*9., a*.7))-.5)*uRise*a : vec3(0.)); }
void main(){
  float ph = seed.x*uPeriod;
  float k = floor((uTime + ph)/uPeriod);
  float age = uTime + ph - k*uPeriod;
  float life = uLife*(.45 + .9*seed.w);
  float h1 = fract(seed.y*31.7 + k*.618), h2 = fract(seed.z*17.3 + k*.414), h3 = fract(seed.w*7.1 + k*.271);
  if(age > life || seed.w > uRate){ gl_Position = vec4(2., 2., 2., 1.); return; }
  float th = h1*6.2832, cz = mix(uUp, 1., h2);
  vec3 dir = vec3(cos(th)*sqrt(1.-cz*cz), cz, sin(th)*sqrt(1.-cz*cz));
  vec3 v = uV + dir*uSpread*(.4 + .8*h3);
  vec3 o = uO + (vec3(h1, h2, h3) - .5)*uBox;
  vec3 p0 = P(age, o, v), p1 = P(max(0., age - uStreak), o, v);
  vec4 c0 = projectionMatrix*viewMatrix*vec4(p0, 1.), c1 = projectionMatrix*viewMatrix*vec4(p1, 1.);
  vec2 s0 = c0.xy/c0.w*uRes, s1 = c1.xy/c1.w*uRes;
  vec2 d = s0 - s1; float L = length(d); d = L > .5 ? d/L : vec2(0., 1.);
  vec2 n = vec2(-d.y, d.x);
  float fpx = projectionMatrix[1][1]*uRes.y;
  vT = uT0*exp(-age*uCool); vC = corner; vA = (1. - age/life)*smoothstep(0., .12, age/life);
#ifdef PLUME
  vec4 vp = viewMatrix*vec4(p0, 1.);
  float sz = uWidth*(1. + uGrow*age);
  vp.xy += vec2(corner.x*2. - 1., corner.y)*sz;
  gl_Position = projectionMatrix*vp;
#else
  float w = min(uWidth*fpx/max(c0.w, .05), 5.) + .8;
  vec4 c = mix(c1, c0, corner.x);
  vec2 sp = mix(s1, s0, corner.x) + n*corner.y*w + d*(corner.x - .5)*w*1.2;
  gl_Position = vec4(sp/uRes*c.w, c.z, c.w);
#endif
}`;
const STREAK_FS = `
uniform float uSoft; varying float vT; varying vec2 vC; varying float vA;
void main(){
  float prof = 1. - abs(vC.y); prof = pow(prof, uSoft);
  float head = mix(.35, 1., vC.x);

const FLOW_VS = `
attribute vec2 corner; attribute vec4 seed;
uniform sampler2D uPath; uniform float uFocus, uN, uNP, uTime, uSpeed, uWidth, uT0, uRate, uLen, uJit, uRev;
uniform vec2 uRes;
varying float vT; varying vec2 vC;
vec3 samp(float pi, float s){ s = clamp(s, 0., 1.)*(uN - 1.); float i = floor(s), f = s - i;
  vec3 a = texture2D(uPath, vec2((i + .5)/uN, (pi + .5)/uNP)).xyz, b = texture2D(uPath, vec2((min(i + 1., uN - 1.) + .5)/uN, (pi + .5)/uNP)).xyz; return mix(a, b, f); }
void main(){
  float pi = fract(seed.z*37.1) < uFocus ? floor(seed.x*2.) : floor(seed.x*uNP); float sp = uSpeed*(.55 + .9*seed.z);
  float s = fract(seed.y + uTime*sp);
  if(seed.w > uRate || s > uRev){ gl_Position = vec4(2., 2., 2., 1.); return; }
  float ph = seed.y*40. + seed.z*13.;
  vec3 off = vec3(sin(ph + uTime*1.3), sin(ph*1.7 + uTime*1.1), cos(ph*1.3 + uTime*.9))*uJit*(.3 + seed.w);
  vec3 p0 = samp(pi, s) + off, p1 = samp(pi, max(0., s - uLen*sp)) + off;
  vec4 c0 = projectionMatrix*viewMatrix*vec4(p0, 1.), c1 = projectionMatrix*viewMatrix*vec4(p1, 1.);
  vec2 s0 = c0.xy/c0.w*uRes, s1 = c1.xy/c1.w*uRes; vec2 d = s0 - s1; float L = length(d); d = L > .5 ? d/L : vec2(0., 1.);
  vec2 n = vec2(-d.y, d.x); float w = uWidth*projectionMatrix[1][1]*uRes.y/max(c0.w, .05) + .7;
  vec4 c = mix(c1, c0, corner.x); vec2 sp2 = mix(s1, s0, corner.x) + n*corner.y*w + d*(corner.x - .5)*w;
  gl_Position = vec4(sp2/uRes*c.w, c.z, c.w);
  vT = uT0*(.75 + .5*fract(seed.z*7.)); vC = corner;
}`;

function lutTex() {
  const pals = [
    [[0, '#07001a'], [0.12, '#1c0350'], [0.28, '#5a078c'], [0.42, '#a01582'], [0.56, '#dc3c3a'], [0.7, '#f4780e'], [0.83, '#fdbd1c'], [0.93, '#fff07a'], [1, '#fffff4']],   // ironbow

    [[0, '#000010'], [0.14, '#1a0c8c'], [0.28, '#0060ff'], [0.42, '#00d0c8'], [0.56, '#30e030'], [0.7, '#f0f000'], [0.84, '#ff6a00'], [0.94, '#ff1a1a'], [1, '#ffffff']], // rainbow
    [[0, '#02061c'], [0.25, '#0b2c78'], [0.5, '#1c7cc8'], [0.7, '#7cc8e8'], [0.82, '#e8c048'], [0.92, '#ffe070'], [1, '#fffff0']],                          // arctic
    [[0, '#000000'], [0.2, '#1c0002'], [0.42, '#5c0008'], [0.62, '#b8081c'], [0.8, '#ff2a2a'], [0.92, '#ff9a8a'], [1, '#fff0ea']],                          // red-black
  ];
  const w = 256, d = new Uint8Array(w * 8 * 4);
  const hex = (h) => [1, 3, 5].map(i => parseInt(h.slice(i, i + 2), 16));
  pals.forEach((st, row) => {
    for (let i = 0; i < w; i++) {
      const x = i / (w - 1); let k = 0; while (k < st.length - 2 && x > st[k + 1][0]) k++;
      const [x0, c0] = st[k], [x1, c1] = st[k + 1], f = clamp((x - x0) / (x1 - x0)), a = hex(c0), b = hex(c1);
      for (let c = 0; c < 3; c++) d[(row * w + i) * 4 + c] = Math.round(lerp(a[c], b[c], f));
      d[(row * w + i) * 4 + 3] = 255;
    }
  });
  const t = new THREE.DataTexture(d, w, 8); t.minFilter = t.magFilter = THREE.LinearFilter; t.needsUpdate = true; return t;
}

const SENSOR = new THREE.ShaderMaterial({ vertexShader: QUAD_VS, depthTest: false, depthWrite: false,

    sharp: { value: 0.55 }, glow: { value: 0.5 }, fpn: { value: 0.035 }, time: { value: 0 }, frame: { value: 0 }, blow: { value: 0 },
    rect: { value: new THREE.Vector4(0, 0, 1, 1) }, texel: { value: new THREE.Vector2(1 / TW, 1 / TH) } },
  fragmentShader: `uniform sampler2D tT, tB, tX, tL; uniform float lo, hi, gam, eq, pal, isoT, isoAmt, heat, outline, flash, dim, uFlat, sharp, glow, fpn, time, frame, blow;
  uniform vec4 rect; uniform vec2 texel; varying vec2 vUv;
  float h12(vec2 p){ vec3 p3 = fract(vec3(p.xyx)*.1031); p3 += dot(p3, p3.yzx+33.33); return fract((p3.x+p3.y)*p3.z); }

  vec3 PI(float n){ return texture2D(tL, vec2(clamp(n, 0., 1.)*(255./256.)+.5/256., .5/8.)).rgb; }   // the words always burn in ironbow
  void main(){
    vec2 uv = vUv;
    vec2 ctr = (rect.xy + rect.zw)*.5, hs = max((rect.zw - rect.xy)*.5, vec2(1e-3));
    vec2 cc = (uv - ctr)/hs*.5;                                   // panel-local, -0.5..0.5
    vec2 uvl = ctr + cc*(1. - .018*dot(cc, cc))*hs*2.;            // a touch of barrel inside each panel
    float T0 = texture2D(tT, uvl).r;
    float Tn = .25*(texture2D(tT, uvl+vec2(texel.x,0.)).r + texture2D(tT, uvl-vec2(texel.x,0.)).r + texture2D(tT, uvl+vec2(0.,texel.y)).r + texture2D(tT, uvl-vec2(0.,texel.y)).r);

    vec2 bl = texture2D(tB, uv).rg;

    float span = max(hi - lo, 1e-4);
    float T = T0 + sharp*(T0 - bl.r)*step(lo - span*.5, bl.r);   // no halo from the cold bezel gap into tight-AGC panels
    float ring = clamp(bl.g*1.6 - m, 0., 1.);
    T = mix(T, lo - span*.12, ring*.9);
    float n = log(1. + max(T - lo, -.2*eq)/eq)/log(1. + span/eq);
    n += glow*max(0., log(1. + max(bl.r - lo, 0.)/eq)/log(1. + span/eq) - .85);
    n = pow(clamp(n, 0., 1.25), gam);
    vec2 fc = gl_FragCoord.xy;

    vec3 c = P(n);
    float iso = step(isoT, T0)*isoAmt*(1. - step(.3, m));
    float hatch = step(.5, fract((fc.x + fc.y + time*90.)/16.));

    c = mix(c, P(.33), uFlat);
    c *= (1. - .5*dot(cc, cc)*2.)*dim;
    c = mix(c, vec3(1., .96, .88), flash*(1. - clamp(ring*3., 0., 1.)));
    // the mind of light (green channel only: no cold ring), then the letters on top
    float fil = clamp(tx.g - tx.r, 0., 1.)*tx.a;

    c = mix(c, ct, smoothstep(.08, .5, m));
    c = mix(c, isoC, ring*outline*.95);
    gl_FragColor = vec4(c, 1.);
  }` });

  const cue = cueAt(beat);
  let heatV = 1.08;
  if (cue) { const age = (beat - cue[0]) * B.spb; drawCue(cue, age); heatV = age < 0.1 ? lerp(0.78, 1.08, age / 0.1) : 1.08; }
  textTex.needsUpdate = true;
  BLUR.uniforms.textT.value = 1.6; blurPass();
  // 4. sensor: the gaps first, then one scissored pass per panel with its own palette and AGC
  const U = SENSOR.uniforms, flash = FLASHES.reduce((f, b) => f + (beat >= b && beat < b + 0.25 ? (b === 52 ? 0.2 : 0.3) * (1 - (beat - b) * 4) : 0), 0);
  U.heat.value = heatV; U.outline.value = cue && cue[4] && cue[4].outline ? clamp((beat - cue[0]) * 4) : 0;
  U.time.value = t; U.frame.value = frame; U.flash.value = flash;
  quad.material = SENSOR; renderer.setRenderTarget(null);
  const setP = (p) => {
    U.lo.value = p ? p.lo : 0; U.hi.value = p ? p.hi : 1.6; U.eq.value = p ? (p.c.eq ?? (p.hi - p.lo) * 0.03) : 0.05; U.pal.value = p ? p.c.pal : 0;
    U.isoT.value = p && p.c.isoT ? p.c.isoT : 99; U.isoAmt.value = p && p.c.iso ? clamp(p.age / 0.15) : 0; U.dim.value = p ? p.dim : 1;
    U.uFlat.value = p && nucOn && b0 > 0 && b0 !== 52 && secAge < 1 / 30 ? 0.35 : 0; U.blow.value = 0;   // frame 0 is the thumbnail: no flat field there
    if (p) U.rect.value.set(p.r.x / W, 1 - (p.r.y + p.r.h) / H, (p.r.x + p.r.w) / W, 1 - p.r.y / H); else U.rect.value.set(0, 0, 1, 1);
  };
  renderer.setScissorTest(false); setP(null); renderer.render(postScene, postCam);
  renderer.setScissorTest(true);
  for (const p of P) { renderer.setScissor(p.r.x, H - p.r.y - p.r.h, p.r.w, p.r.h); setP(p); renderer.render(postScene, postCam); }
  renderer.setScissorTest(false);
  ctx.drawImage(renderer.domElement, 0, 0);
  // 5. bezels and readouts
  bezels(P, one);
  hud(P, t, beat, one);
  finish.apply(ctx, frame, { grain: 0.05, vignette: 0.3 });
};
