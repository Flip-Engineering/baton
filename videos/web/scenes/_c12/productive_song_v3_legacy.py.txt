# ================================================================ piece-owned voices (§4.5: built here)
def _sq(f, t, ph=0.0, top=12000.0):
    """Band-limited square (odd harmonics below `top`): no aliasing on the high cowbell."""
    y = np.zeros_like(t)
    k = 1
    while k * f < top:
        y += np.sin(2 * np.pi * k * f * t + k * ph) / k
        k += 2
    return y * (4 / np.pi)


def pf_kick():
    """Phonk kick: a clipped sine swept 180 -> 45 Hz in ~40 ms, a saturated 350 ms tail, a rectified 2nd-harmonic
    layer (80-320 Hz) and a 1.8-5.5 kHz beater click, so the downbeat reads on a phone."""
    t = _t(0.42)
    f = 45 + 135 * np.exp(-t / 0.011)
    x = np.sin(2 * np.pi * np.cumsum(f) / SR)
    env = np.where(t < 0.2, 1.0, np.exp(-(t - 0.2) * 14)) * np.clip((0.42 - t) / 0.03, 0, 1)
    body = np.tanh(x * 5.0) * env
    r = np.abs(body)
    h2 = bp(r - r.mean(), 80, 320, 2) * 1.1
    click = bp(_R.standard_normal(len(t)), 1800, 5500) * np.exp(-t * 380) * 0.9
    y = np.tanh((body * 0.9 + h2 + click) * 1.3) * 0.92
    return _fade(y, 0.0, 0.01)


def pf_808(n, dur, glide_to=None, glide_ms=80, glide_at=None, drive=5.0):
    """Distorted phonk 808: a sine with a short pitch punch, smoothstep glides (the flat-2 slide), clipped hard
    (odd harmonics) plus a rectified 2nd-harmonic layer (90-480 Hz) so the line reads on phones."""
    dur = max(dur, 0.12)
    t = _t(dur + 0.03)
    f = note_hz(n) * (1 + 0.8 * np.exp(-t * 45))
    if glide_to is not None:
        g = glide_ms / 1000
        t0 = max(0.0, dur - g) if glide_at is None else glide_at
        u = np.clip((t - t0) / max(g, 1e-3), 0, 1)
        f = f * 2 ** ((glide_to - n) * (u * u * (3 - 2 * u)) / 12)
    x = np.sin(2 * np.pi * np.cumsum(f) / SR)
    env = np.exp(-t * 0.9) * np.minimum(1, t / 0.003) * np.clip((dur + 0.03 - t) / 0.03, 0, 1)
    body = np.tanh(x * drive) * env
    r = np.abs(body)
    h2 = bp(r - r.mean(), 90, 480, 2) * 0.9
    return np.tanh((body + h2) * 1.2) * 0.8


def pf_cow(n, dur=0.3, drive=2.4):
    """Tuned phonk cowbell: the 808 cowbell's two squares (f and 1.48 f, band-limited), band-passed 380-6500 Hz,
    a two-stage envelope (the clack, then the ring) and drive: the doomsday bell."""
    L = max(dur, 0.24) + 0.06
    t = _t(L)
    f = note_hz(n)
    x = _sq(f, t) + 0.8 * _sq(f * 1.48, t, 0.7)
    x = bp(x, 380, 6500, 2)
    env = (0.62 * np.exp(-t * 30) + 0.38 * np.exp(-t * 6.5)) * np.minimum(1, t / 0.0015)
    y = np.tanh(x * env * drive) / np.tanh(drive)
    return _fade(y * 0.5, 0.0, 0.03)


def pf_stab(chord, dur=0.9):
    """The distorted cowbell-chord stab (WORKER RULE, the drop-B callbacks): the chord's cowbells summed BEFORE the
    drive, so they intermodulate into one furnace-bell; a long ring."""
    t = _t(dur + 0.2)
    x = np.zeros_like(t)
    for n in chord:
        f = note_hz(n)
        x += _sq(f, t, _R.random()) + 0.8 * _sq(f * 1.48, t, _R.random())
    x = bp(x / len(chord), 300, 7000, 2)
    env = (0.5 * np.exp(-t * 18) + 0.5 * np.exp(-t * 3.2)) * np.minimum(1, t / 0.002)
    y = lp(np.tanh(x * env * 3.0) * 0.5, 5500)       # band-limited after the drive: a clipped square chord overshoots AAC by 4 dB
    return _fade(y, 0.0, 0.08)


def pf_press(heavy=False):
    """Press stamp for the piston 8ths: a 70 Hz thump (60 ms), an inharmonic clang (487-3097 Hz, 90 ms) and an air
    hiss (3-9 kHz, 25 ms late). 110 ms, so it fits a 16th. heavy: the slam (45 Hz, longer clang, air release)."""
    d = 0.55 if heavy else 0.11
    t = _t(d)
    f0 = 45 if heavy else 70
    thump = np.sin(2 * np.pi * np.cumsum(f0 * (1 + 0.9 * np.exp(-t * 60))) / SR) * np.exp(-t * (9 if heavy else 45))
    parts = ((487, 1.0), (1123, 0.8), (1789, 0.6), (2411, 0.45), (3097, 0.3))
    clang = sum(a * np.sin(2 * np.pi * f * t + i) for i, (f, a) in enumerate(parts)) / 3.0 * np.exp(-t * (14 if heavy else 38))
    air = bp(_R.standard_normal(len(t)), 3000, 9000) * np.where(t > 0.025, np.exp(-(t - 0.025) * (7 if heavy else 50)), 0)
    y = np.tanh((thump * (1.4 if heavy else 0.9) + clang * 0.8 + air * 0.5) * 1.4) * (0.9 if heavy else 0.7)
    return _fade(y, 0.0, 0.02 if heavy else 0.01)


def pf_clank():
    """Clank snare: a noise snare (1.2-9 kHz, 170 ms), a 190 Hz body, a struck-steel layer (873-3571 Hz, 260 ms)
    and a crack: the shop floor's backbeat."""
    t = _t(0.36)
    nz = bp(_R.standard_normal(len(t)), 1200, 9000, 2) * np.exp(-t * 18)
    body = np.sin(2 * np.pi * np.cumsum(190 * (1 + 0.5 * np.exp(-t * 80))) / SR) * np.exp(-t * 30)
    metal = sum(a * np.sin(2 * np.pi * f * t + a) for f, a in ((873, 1.0), (1391, 0.8), (2207, 0.6), (2903, 0.45), (3571, 0.3)))
    metal = metal / 3 * np.exp(-t * 11)
    crack = lp(_R.standard_normal(len(t)), 7500) * np.exp(-t * 600) * 0.9      # band-limited: a raw 1 ms crack overshoots AAC
    y = np.tanh((nz * 0.9 + body * 0.7 + metal * 0.55 + crack) * 1.5) * 0.85
    return _fade(y, 0.0, 0.02)


def pf_hat(open_=False):
    """Steam-hiss hat: 6-12 kHz hiss with a faint whistle partial (8.4 kHz); closed 45 ms, open 230 ms."""
    d = 0.23 if open_ else 0.045
    t = _t(d)
    n = bp(_R.standard_normal(len(t)), 6000, 12000, 2)
    wh = np.sin(2 * np.pi * 8400 * t + _R.random() * 6) * 0.12
    env = np.minimum(1, t / 0.003) * np.exp(-t / (d / 3.2))
    return (n + wh) * env * 0.55


def pf_furnace(dur, lpf=None, shape=None, seed=0):
    """Furnace roar bed (stereo, decorrelated): noise band-passed 60-900 Hz, a 0.25 Hz breath, a 3-7 Hz flicker and
    sparse crackle. lpf=(a, b): a low-pass opening a -> b Hz across the event (the break).
    shape 'tail': after the closer the roar dies away, then swells back over the last 2 beats across the seam."""
    t = _t(dur)
    rng = np.random.default_rng(1538 + seed)
    out = np.zeros((len(t), 2))
    for c in range(2):
        base = bp(rng.standard_normal(len(t)), 60, 900, 2)
        base /= np.std(base) + 1e-9
        fl = lp(rng.standard_normal(len(t)), 6)
        fl /= np.std(fl) + 1e-9
        g = np.clip(1 + 0.25 * np.sin(2 * np.pi * 0.25 * t + 1.3 + c) + 0.22 * fl, 0.25, 2.0)
        cr = np.zeros(len(t))
        k = int(35 * dur)
        cr[rng.integers(0, len(t), k)] = rng.uniform(-1, 1, k)
        cr = hp(cr, 2000) * 6
        out[:, c] = np.tanh(base * g * 0.45 + cr) * 0.6
    if lpf is not None:
        cut = np.exp(np.linspace(np.log(lpf[0]), np.log(lpf[1]), len(t)))
        out = audio.tv_filter(out, cut, "low", 0.9)
    if shape == "tail":
        rise = np.clip((t - (dur - 2 * SPB - 0.3)) / (2 * SPB), 0, 1)       # the last 2 beats before the seam
        env = np.exp(-t * 2.2) * 0.6 + 1.3 * rise ** 2
        env *= np.clip((dur - t) / 0.08, 0, 1)
        out = (out.T * env).T
    return _fade(out, 0.02, 0.06)


def pf_door():
    """Furnace-door blast (frame 0, the sigil, the closer): the iron door's clunk (low inharmonic modes + latch),
    a fire whoosh (200-4000 Hz, 8 ms attack, flicker) and a 55 -> 32 Hz boom."""
    t = _t(1.6)
    rng = np.random.default_rng(60)
    clunk = sum(a * np.sin(2 * np.pi * f * t + p) * np.exp(-t * d)
                for f, a, d, p in ((96, 1.0, 9, 0), (231, 0.7, 12, 1), (407, 0.5, 16, 2), (683, 0.35, 20, 3), (1130, 0.25, 28, 4)))
    latch = bp(rng.standard_normal(len(t)), 2000, 8000) * np.exp(-t * 300)
    fl = 1 + 0.35 * lp(rng.standard_normal(len(t)), 9) / 0.02
    fire = np.stack([bp(rng.standard_normal(len(t)), 200, 4000) for _ in range(2)], 1)
    fire = (fire.T * (np.minimum(1, t / 0.008) * np.exp(-t * 2.6) * np.clip(fl, 0.3, 2.0))).T
    boom = np.sin(2 * np.pi * np.cumsum(32 + 23 * np.exp(-t * 6)) / SR) * np.exp(-t * 2.5)
    mono = clunk * 0.6 + latch * 0.8 + boom * 1.2
    y = np.tanh(fire * 1.1 + mono[:, None]) * 0.85
    return _fade(y, 0.0, 0.1)


def pf_hum(dur, surge=False):
    """Grid hum: 60 Hz with 180, 300 and 420 Hz (the transformer's odd series) and a rectified 120 Hz buzz in the
    400-4000 Hz band (what a phone hears). surge: a fast swell that sags back (POWER)."""
    t = _t(dur)
    y = sum(a * np.sin(2 * np.pi * f * t) for f, a in ((60, 1.0), (180, 0.55), (300, 0.35), (420, 0.25)))
    buzz = bp(np.abs(np.sin(2 * np.pi * 60 * t)) ** 8, 400, 4000, 2) * 2.0
    y = (y * 0.3 + buzz * 0.45)
    if surge:
        env = np.minimum(1, t / 0.05) * (0.45 + 0.55 * np.exp(-t * 1.6))
    else:
        env = np.minimum(1, t / 0.15)
    return _fade(np.tanh(y * env * 1.6) * 0.6, 0.01, 0.12)


def pf_zap():
    """Transformer arc zap: a crackling burst, the hum's harmonics and an 8 kHz -> 500 Hz chirp."""
    return _fade(audio.arc_zap(0.6)[: int(0.3 * SR)], 0.0, 0.08)


def pf_whistle(dur):
    """Factory shift whistle: a 3-pipe steam chord G5-Bb5-D6 (sine + a little 2nd harmonic, slow vibrato), breath
    noise at each pipe, 30 ms attack, a -30 cent sag at the end."""
    t = _t(dur)
    sag = 2 ** (-30 / 1200 * np.clip((t - (dur - 0.35)) / 0.35, 0, 1))
    rng = np.random.default_rng(33)
    y = np.zeros_like(t)
    for m in (G5, Bb5, D6):
        f = note_hz(m) * sag * (1 + 0.003 * np.sin(2 * np.pi * rng.uniform(4.5, 6) * t))
        ph = 2 * np.pi * np.cumsum(f) / SR
        y += np.sin(ph) + 0.18 * np.sin(2 * ph)
        y += bp(rng.standard_normal(len(t)), note_hz(m) * 0.9, note_hz(m) * 1.12, 2) * 1.2
    br = bp(rng.standard_normal(len(t)), 1500, 7000) * 0.12
    env = np.minimum(1, t / 0.03) * np.clip((dur - t) / 0.12, 0, 1)
    return np.tanh((y / 3 + br) * env * 1.8) * 0.5


def pf_crush():
    """CHIPS: the cowbell stuttered on 32nds (G5 G5 Ab5 G5 D6 G6), bit-crushed to 4 bits at 5.5 kHz."""
    s32 = STEP / 2
    notes = (G5, G5, Ab5, G5, D6, G5 + 12)
    out = np.zeros(int((len(notes) * s32 + 0.25) * SR))
    for k, n in enumerate(notes):
        c = pf_cow(n, 0.12, 3.0)[: int(s32 * SR * 0.9)]
        c = _fade(c, 0.0, 0.004)
        i = int(k * s32 * SR)
        out[i:i + len(c)] += c
    return bitcrush(out * 0.9, 4, 5500)


def pf_girder():
    """Steel-girder clang (STEEL): a struck I-beam's free-free bending modes (1 : 2.76 : 5.40 : 8.93 : 13.3 on
    118 Hz, plus two torsional modes), a hard strike transient, 1.1 s ring."""
    t = _t(1.3)
    f0 = 118.0
    modes = [(1.0, 1.0, 4.5), (2.76, 0.9, 5.5), (5.40, 0.7, 7), (8.93, 0.5, 9), (13.34, 0.35, 12), (4.1, 0.4, 6), (7.3, 0.3, 8)]
    y = sum(a * np.sin(2 * np.pi * f0 * r * t + r) * np.exp(-t * d) for r, a, d in modes)
    strike = bp(_R.standard_normal(len(t)), 1500, 9000) * np.exp(-t * 160)
    return _fade(np.tanh((y * 0.5 + strike) * 1.6) * 0.8, 0.0, 0.1)


def pf_chuff():
    """TRAINS: four steam chuffs on 32nds, accelerating out of the piston (steam hiss, 400-5000 Hz)."""
    offs = (0.0, 0.055, 0.1, 0.14)
    out = np.zeros(int(0.45 * SR))
    for k, o in enumerate(offs):
        t = _t(0.12)
        c = bp(_R.standard_normal(len(t)), 400, 5000) * np.exp(-t * 30) * (0.7 + 0.1 * k)
        i = int(o * SR)
        out[i:i + len(c)] += c
    return np.tanh(out * 1.3) * 0.7


def pf_revriff(dur):
    """The break's reversed cowbell riff: RIFF_A rendered wet (hall), reversed, so it swells toward the silence."""
    L = int((dur + 1.0) * SR)
    x = np.zeros(L)
    for k, n in enumerate(RIFF_A + RIFF_A):
        c = pf_cow(n, 0.3, 2.0)
        i = int(k * 2 * STEP * SR)
        if i < L:
            x[i:i + len(c)] += c[: L - i]
    wet = reverb(x, "hall", hp_hz=300)
    y = (np.stack([x, x], 1) * 0.5 + wet * 1.4)[::-1]
    y = y[-int(dur * SR):]
    return _fade(y / (np.abs(y).max() + 1e-9) * 0.6, 0.2, 0.01)


def pf_hammer():
    """Drop-forge hammer (the piece's own impact, never the shared one): a 98 -> 28 Hz boom, the steel slab's
    low inharmonic thud modes (143/331/577/911 Hz), a strike crack and a steam-release hiss 80 ms after."""
    t = _t(2.0)
    rng = np.random.default_rng(7)
    boom = np.sin(2 * np.pi * np.cumsum(28 + 70 * np.exp(-t * 9)) / SR) * np.exp(-t * 2.4)
    slab = sum(a * np.sin(2 * np.pi * f * t + i) * np.exp(-t * d)
               for i, (f, a, d) in enumerate(((143, 1.0, 14), (331, 0.7, 18), (577, 0.5, 22), (911, 0.35, 30))))
    crack = bp(rng.standard_normal(len(t)), 1000, 6000) * np.exp(-t * 120)
    steam = bp(rng.standard_normal(len(t)), 2000, 8000) * np.where(t > 0.08, np.exp(-(t - 0.08) * 3.0), 0) * 0.35
    y = np.tanh((boom * 1.3 + slab * 0.55 + crack * 0.7 + steam) * 1.8) * 0.9
    return _fade(y, 0.0, 0.1)


def pf_sheet(dur=2.0):
    """Thunder sheet (the piece's own crash): a struck steel sheet, a wobbling 400-6000 Hz noise band (the
    sheet flexing at 7 -> 3 Hz), 20 inharmonic plate partials (700-5500 Hz) and a thin 5 kHz sheen."""
    t = _t(dur)
    rng = np.random.default_rng(int(dur * 1000) + 11)
    wob = 1 + 0.55 * np.sin(2 * np.pi * np.cumsum(7 - 4 * np.clip(t / dur, 0, 1)) / SR)
    band = bp(rng.standard_normal(len(t)), 400, 6000) * wob * np.exp(-t * 2.4)
    parts = sum(np.sin(2 * np.pi * f * t * (1 + 0.004 * np.sin(2 * np.pi * 5 * t)) + rng.random() * 6)
                * np.exp(-t * rng.uniform(1.8, 3.5)) for f in rng.uniform(700, 5500, 20)) / 20
    sheen = hp(rng.standard_normal(len(t)), 5000) * np.exp(-t * 4) * 0.4
    y = np.tanh((band * 0.9 + parts * 2.2 + sheen) * 1.4) * np.minimum(1, t / 0.002) * 0.5
    return _fade(y, 0.0, 0.05)


def pf_vent(dur=1.5):
    """Steam-vent riser (the piece's own reverse cymbal): pressurised hiss sweeping 1.5 -> 8 kHz, rising to a
    hard stop on its step, a faint valve whistle climbing with it."""
    t = _t(dur)
    u = t / dur
    rng = np.random.default_rng(88)
    y = audio.tv_filter(rng.standard_normal(len(t)), 1500 * (8000 / 1500) ** u, "band", q=1.4)
    wh = np.sin(2 * np.pi * np.cumsum(900 * (1 + 1.2 * u)) / SR) * 0.08
    return _fade((y + wh) * u ** 1.8 * 0.55, 0.01, 0.002)


def pf_boom(dur=1.2):
    """Furnace sub (the piece's own sub drop): a 55 -> 26 Hz sweep with a 40-140 Hz fire rumble under it."""
    t = _t(dur)
    rng = np.random.default_rng(26)
    f = 55 * (26 / 55) ** (t / dur)
    s = np.sin(2 * np.pi * np.cumsum(f) / SR)

# ================================================================ wordless gang hits and the camera's shutter
def pf_gang(big=False):
    """The crowd without words (client, 27 Sep: no synthesized word vocals): a stomp stack (a 62 Hz floor thud and
    boot-scuff noise from ~24 bodies, spread over 18 ms) and a clap stack (24 hands, 900-5200 Hz bursts, spread over
    22 ms), both in a big room. big: 40 bodies, a longer room tail and a low 'hey'-less chest thump."""
    rng = np.random.default_rng(404 if big else 403)
    n = 40 if big else 24
    L = 0.9 if big else 0.6
    t = _t(L)
    out = np.zeros((len(t), 2))
    for k in range(n):
        d = int(abs(rng.normal(0, 0.009 if big else 0.007)) * SR)
        c = k % 2
        tt = t[: len(t) - d]
        thud = np.sin(2 * np.pi * np.cumsum(62 * (1 + 0.6 * np.exp(-tt * 70))) / SR) * np.exp(-tt * 22)
        scuff = bp(rng.standard_normal(len(tt)), 300, 2500) * np.exp(-tt * 60) * 0.35
        clap = bp(rng.standard_normal(len(tt)), 900 * rng.uniform(0.85, 1.15), 5200, 2) * np.exp(-tt * rng.uniform(55, 80))
        dc = int(abs(rng.normal(0.0, 0.011)) * SR)
        cl = np.zeros(len(tt))
        cl[dc:] = clap[: len(tt) - dc]
        out[d:, c] += thud * 0.9 + scuff + cl * 0.8
    out /= n ** 0.75
    wet = reverb(out.mean(1), "large_hall", hp_hz=250)
    y = np.tanh((out + wet[: len(out)] * (0.6 if big else 0.45)) * 1.6) * 0.8
    return _fade(y, 0.0, 0.08)


def pf_nuc():
    """The thermal camera's NUC shutter: a solenoid snap (a 1 ms click), the flag hitting its stop (3.1/4.7/6.3 kHz
    ringing, 25 ms) and the return tick 60 ms later. The piece's signature sound, on every wall freeze."""
    t = _t(0.12)
    rng = np.random.default_rng(55)
    click = lp(rng.standard_normal(len(t)), 7000) * np.exp(-t / 0.0015) * 1.6    # band-limited: a raw 1 ms impulse overshoots the AAC encode
    ring = sum(a * np.sin(2 * np.pi * f * t) for f, a in ((3100, 1.0), (4700, 0.6), (6300, 0.4))) * np.exp(-t * 160)
    back = lp(np.where(t > 0.06, rng.standard_normal(len(t)) * np.exp(-(t - 0.06) / 0.0012), 0), 7000) * 0.8
    body = np.sin(2 * np.pi * 180 * t) * np.exp(-t * 90) * 0.5
    return _fade(np.tanh((click + ring * 0.5 + back + body) * 1.4) * 0.7, 0.0, 0.01)

# ================================================================ the score (32nd-note grid)
class Score:
    def __init__(self):
        self.bars = [{} for _ in range(NB)]

    @staticmethod
    def at(beat):
        b = int(beat // 4 + 1e-9)
        return b, int(round((beat - 4 * b) * 8))

    def hit(self, track, beat, ch="x"):
        b, i = self.at(beat)
        s = self.bars[b].get(track, "." * 32)
        self.bars[b][track] = s[:i] + ch + s[i + 1:]

    def note(self, track, beat, n, beats, **o):
        b, i = self.at(beat)
        lst = self.bars[b].setdefault(track, [None] * 32)
        lst[i] = (n, beats * 8, o) if o else (n, beats * 8)

    def clear(self, tracks, b0, b1):
        """Remove every hit / note of `tracks` with b0 <= beat < b1."""
        for tr in tracks:
            for b in range(NB):
                v = self.bars[b].get(tr)
                if v is None:
                    continue
                for i in range(32):
                    beat = 4 * b + i / 8
                    if b0 - 1e-9 <= beat < b1 - 1e-9:
                        if isinstance(v, str):
                            v = v[:i] + "." + v[i + 1:]
                        else:
                            v[i] = None
                self.bars[b][tr] = v

    def stem(self, beat, arr, onset, text=None, gain=1.0):
        b = int(beat // 4 + 1e-9)
        self.bars[b].setdefault("_stem", []).append((arr, (beat - 4 * b) * 4, gain, text, onset))

    def mark(self, beat):
        """A textless shout hit for a word inside a line (so hits.json carries every shouted word)."""
        self.stem(beat, np.zeros((64, 2)), 0.0, None, 1.0)

    def set(self, bar, key, val):
        self.bars[bar][key] = val
